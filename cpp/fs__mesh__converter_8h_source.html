37inline Mesh convert_fs_mesh(const fs::Mesh &fs_mesh) {
38 Mesh out;
39 size_t nv = fs_mesh.num_vertices();
40 out.vertices.reserve(nv);
41 for (size_t i = 0; i < nv; i++) {
42 out.vertices.push_back(Vec3(
43 fs_mesh.vertices[i * 3],
44 fs_mesh.vertices[i * 3 + 1],
45 fs_mesh.vertices[i * 3 + 2]));
46 }
47 size_t nf = fs_mesh.num_faces();
48 out.triangles.reserve(nf);
49 for (size_t i = 0; i < nf; i++) {
50 out.triangles.push_back(Triangle{
51 static_cast<uint32_t>(fs_mesh.faces[i * 3]),
52 static_cast<uint32_t>(fs_mesh.faces[i * 3 + 1]),
53 static_cast<uint32_t>(fs_mesh.faces[i * 3 + 2])});
54 }
55 return out;
56}

148inline Mesh convert_fs_mesh(const fs::Mesh &fs_mesh,
149 const std::vector<float> &morph_data,
150 const std::vector<uint8_t> &rgb_colors,
151 float nan_alpha = 1.0f) {
152 Mesh out = convert_fs_mesh(fs_mesh);
153 size_t nv = out.vertices.size();
154 out.colors.reserve(nv);
155 for (size_t i = 0; i < nv; i++) {
156 if (std::isnan(morph_data[i])) {
157 out.colors.push_back(Color(1.0f, 1.0f, 1.0f, nan_alpha));
158 } else {
159 out.colors.push_back(Color(
160 rgb_colors[i * 3] / 255.0f,
161 rgb_colors[i * 3 + 1] / 255.0f,
162 rgb_colors[i * 3 + 2] / 255.0f,
163 1.0f));
164 }
165 }
166 out.update_transparency();
167 return out;
