371 void compute_bounding_box(Vec3 &min_bound, Vec3 &max_bound) const {
372 bool first = true;
373
374 // Add the 8 transformed corners of a box to the bounds.
375 auto add_box = [&](const Vec3 &box_min, const Vec3 &box_max, const Mat4 &m) {
376 for (int c = 0; c < 8; ++c) {
377 Vec3 corner(
378 (c & 1) ? box_max.x : box_min.x,
379 (c & 2) ? box_max.y : box_min.y,
380 (c & 4) ? box_max.z : box_min.z);
381 Vec3 t = transform_point(m, corner);
382 if (first) {
383 min_bound = t;
384 max_bound = t;
385 first = false;
386 } else {
387 min_bound = glm::min(min_bound, t);
388 max_bound = glm::max(max_bound, t);
389 }
390 }
391 };
392
393 for (size_t i = 0; i < meshes.size(); ++i) {
394 const Mesh &mesh = meshes[i];
395 if (mesh.vertices.empty())
396 continue;
397 Vec3 mesh_min, mesh_max;
398 mesh.compute_bounding_box(mesh_min, mesh_max);
399 add_box(mesh_min, mesh_max, transform(i));
400 }
401
402 // Line layers are content unless they opted out, see
403 // LineLayer::affects_bounds.
404 const bool framed_by_meshes = !first;
405 for (size_t i = 0; i < lines.size(); ++i) {
406 if (!lines[i].affects_bounds)
407 continue;
408 Vec3 line_min, line_max;
409 if (lines[i].compute_bounding_box(line_min, line_max))
410 add_box(line_min, line_max, line_transform(i));
411 }
412
413 // A scene without meshes and without any layer that claims to define the
414 // extent: frame it with the lines anyway (they can only be decoration
415 // if there is nothing else), because otherwise the scene would have no
416 // geometry to derive a camera from.
417 if (!framed_by_meshes && first) {
418 for (size_t i = 0; i < lines.size(); ++i) {
419 Vec3 line_min, line_max;
420 if (lines[i].compute_bounding_box(line_min, line_max))
421 add_box(line_min, line_max, line_transform(i));
422 }
423 }
424
425 if (first) { // scene empty or all meshes empty
426 min_bound = Vec3(0.0f);
427 max_bound = Vec3(0.0f);
428 }
429 }
