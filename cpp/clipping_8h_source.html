85inline ClipPlane clip_plane_to_view_space(const ClipPlane &plane,
86 const Vec3 &eye,
87 const Mat4 &view) {
88 ClipPlane out;
89 out.space = plane.space;
90
91 float len = glm::length(plane.normal);
92 if (len < 1e-12f) {
93 // Degenerate normal: keep everything (no clipping) instead of
94 // discarding the whole scene.
95 out.normal = Vec3(0.0f, 0.0f, 0.0f);
96 out.offset = 0.0f;
97 return out;
98 }
99
100 Vec3 n = plane.normal / len;
101 out.normal = transform_direction(view, n);
102 if (plane.space == PlaneSpace::WORLD) {
103 // Shift the plane constant so that the cut stays at its world
104 // position (see the derivation above).
105 out.offset = glm::dot(n, eye) + plane.offset;
106 } else {
107 out.offset = plane.offset;
108 }
109 return out;
110}
